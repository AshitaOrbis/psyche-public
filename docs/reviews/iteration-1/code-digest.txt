  instrument: Instrument,
  score: ScoringFn
): void {
  registry.set(instrument.id, { instrument, score });
}

export function getInstrument(id: string): RegisteredInstrument | undefined {
  return registry.get(id);
}

export function getAllInstruments(): RegisteredInstrument[] {
  return Array.from(registry.values());
}

  return Array.from(registry.keys());
}

  const reg = registry.get(id);
  return reg ? "adaptive" in reg.instrument && (reg.instrument as Record<string, unknown>).adaptive === true : false;
}

 * Get the list of instrument IDs for a given tier.
 *
 * Accumulates instruments from lite through the target tier,
 * applying replacement rules at each level.
 */
export function getInstrumentsForTier(tier: Tier): string[] {
  const instruments = new Set<string>();
  const targetIdx = TIER_ORDER.indexOf(tier);

  for (let i = 0; i <= targetIdx; i++) {
    const def = TIER_DEFS[TIER_ORDER[i]!]!;

    // Apply replacements: remove old instruments
    for (const oldId of Object.keys(def.replaces)) {
      instruments.delete(oldId);
    }

    // Add new instruments
    for (const id of def.instruments) {
      instruments.add(id);
    }
  }

  return Array.from(instruments);
}

import type {
  Instrument,
  InstrumentSession,
  InstrumentResult,
  ScaleScore,
  Item,
  ItemResponse,
} from "../instruments/types";
import type { CATSession } from "./cat-controller";

/**
 * Generic binary (True/False) scoring engine.
 * True = 1, False = 0; reverse scoring supported.
 */
export function scoreBinary(
  instrument: Instrument,
  session: InstrumentSession
): InstrumentResult {
  const responseMap = new Map<string, ItemResponse>();
  for (const r of session.responses) {
    responseMap.set(r.itemId, r);
  }

  const scaleItems = new Map<string, { item: Item; value: number }[]>();

  for (const item of instrument.items) {
    const resp = responseMap.get(item.id);
    if (!resp || typeof resp.value !== "number") continue;

    let value = resp.value; // 1 = first label (True), 0 = second label (False)
    if (item.reversed) {
      value = 1 - value;
    }

    const existing = scaleItems.get(item.scaleId) ?? [];
    existing.push({ item, value });
    scaleItems.set(item.scaleId, existing);
  }

  const scores: ScaleScore[] = [];
  for (const scale of instrument.scales) {
    const items = scaleItems.get(scale.id);
    if (!items || items.length === 0) continue;

    const raw = items.reduce((sum, i) => sum + i.value, 0) / items.length;
    const normalized = raw * 100; // 0-1 → 0-100

    scores.push({
      scaleId: scale.id,
      scaleName: scale.name,
      raw,
      normalized,
      itemCount: items.length,
    });
  }

  return {
    instrumentId: instrument.id,
    completedAt: session.completedAt ?? Date.now(),
    scores,
  };
}

/**
 * Generic Likert scoring engine.
 * Handles reverse scoring, scale aggregation, and normalization.
 */
export function scoreLikert(
  instrument: Instrument,
  session: InstrumentSession
): InstrumentResult {
  const responseMap = new Map<string, ItemResponse>();
  for (const r of session.responses) {
    responseMap.set(r.itemId, r);
  }

  const scaleItems = new Map<string, { item: Item; value: number }[]>();

  for (const item of instrument.items) {
    const resp = responseMap.get(item.id);
    if (!resp || typeof resp.value !== "number") continue;

    let value = resp.value;
    if (item.reversed && item.response.type === "likert") {
      const { min, max } = item.response;
      value = max + min - value;
    }

    const existing = scaleItems.get(item.scaleId) ?? [];
    existing.push({ item, value });
    scaleItems.set(item.scaleId, existing);
  }

  const scores: ScaleScore[] = [];
  for (const scale of instrument.scales) {
    const items = scaleItems.get(scale.id);
    if (!items || items.length === 0) continue;

    const raw = items.reduce((sum, i) => sum + i.value, 0) / items.length;

    // Normalize assuming all items share the same response format
    const firstItem = instrument.items.find((i) => i.scaleId === scale.id);
    let normalized = raw;
    if (firstItem?.response.type === "likert") {
      const { min, max } = firstItem.response;
      normalized = ((raw - min) / (max - min)) * 100;
    }

    scores.push({
      scaleId: scale.id,
      scaleName: scale.name,
      raw,
      normalized,
      itemCount: items.length,
    });
  }

  return {
    instrumentId: instrument.id,
    completedAt: session.completedAt ?? Date.now(),
    scores,
  };
}

/**
 * Convert a completed CAT session into an InstrumentResult.
 * Theta estimates are converted to 0-100 percentile scores via normal CDF.
 */
export function scoreAdaptive(
  instrumentId: string,
  catSession: CATSession,
  scaleNames?: Map<string, string>,
): InstrumentResult {
  const scores = catSessionToScores(catSession);
  // Map scale names if provided
  if (scaleNames) {
    for (const score of scores) {
      score.scaleName = scaleNames.get(score.scaleId) ?? score.scaleId;
    }
  }
  return {
    instrumentId,
    completedAt: Date.now(),
    scores,
  };
}

========== web/src/scoring/grm-engine.ts ==========
/**
 * Graded Response Model (GRM) engine for Computerized Adaptive Testing.
 *
 * Implements Samejima's (1969) GRM for polytomous items (e.g., 5-point Likert).
 * Uses Expected A Posteriori (EAP) estimation with Gauss-Hermite quadrature.
 *
 * Mathematical basis:
 *   P*(k|θ) = 1 / (1 + exp(-a * (θ - b_k)))
 *   P(X=k|θ) = P*(k|θ) - P*(k+1|θ)  where P*(0)=1, P*(K)=0
 *   I(θ) = a² * Σ_k [P'(X=k|θ)² / P(X=k|θ)]
 */

/** IRT parameters for a single GRM item. */
export interface GRMItem {
  /** Unique item identifier */
  id: string;
  /** Item text for display */
  text: string;
  /** Discrimination parameter (a > 0) */
  discrimination: number;
  /** Threshold parameters [b₁, b₂, ..., b_{K-1}], strictly ordered */
  thresholds: number[];
  /** Number of response categories (e.g., 5 for 1-5 Likert) */
  numCategories: number;
  /** Dimension/facet this item measures */
  dimensionId: string;
  /** Whether to reverse-score this item */
  reverse?: boolean;
}

/** Response to a CAT item. */
export interface CATResponse {
  itemId: string;
  /** 0-indexed category (0 = lowest, numCategories-1 = highest) */
  category: number;
}

/** Quadrature point for EAP estimation. */
interface QuadPoint {
  theta: number;
  weight: number;
}

// ── Constants ──────────────────────────────────────────────────

const CLAMP_MIN = 1e-10;
const CLAMP_MAX = 1 - 1e-10;
const QUAD_RANGE = 4.0; // ±4 SD
const QUAD_POINTS = 41;  // Odd for symmetry

// ── Core GRM Functions ─────────────────────────────────────────

/**
 * Cumulative probability P*(k|θ) = 1 / (1 + exp(-a*(θ - b_k))).
 * P*(0) = 1, P*(K) = 0 by convention.
 */
function cumulativeProb(item: GRMItem, theta: number, k: number): number {
  if (k <= 0) return 1.0;
  if (k >= item.numCategories) return 0.0;
  const threshold = item.thresholds[k - 1];
  if (threshold === undefined) return 0.0;
  const z = item.discrimination * (theta - threshold);
  // Numerically stable logistic
  if (z > 30) return 1.0;
  if (z < -30) return 0.0;
  return 1.0 / (1.0 + Math.exp(-z));
}

/**
 * Category probability P(X=k|θ) = P*(k) - P*(k+1).
 * Returns a clamped value in [CLAMP_MIN, CLAMP_MAX].
 */
export function categoryProbability(item: GRMItem, theta: number, category: number): number {
  const pStar = cumulativeProb(item, theta, category);
  const pStarNext = cumulativeProb(item, theta, category + 1);
  return Math.max(CLAMP_MIN, Math.min(CLAMP_MAX, pStar - pStarNext));
}

/**
 * All category probabilities for an item. Normalized to sum to 1.
 */
export function categoryProbabilities(item: GRMItem, theta: number): number[] {
  const probs: number[] = [];
  let sum = 0;
  for (let k = 0; k < item.numCategories; k++) {
    const p = categoryProbability(item, theta, k);
    probs.push(p);
    sum += p;
  }
  // Normalize
  if (sum > 0) {
    for (let i = 0; i < probs.length; i++) {
      probs[i] = probs[i]! / sum;
    }
  }
  return probs;
}

/**
 * Fisher information I(θ) for a single item.
 * Uses numerical approximation via category probabilities.
 *
 * I(θ) = a² * Σ_k [(P*_k(1-P*_k) - P*_{k+1}(1-P*_{k+1}))² / P(X=k)]
 */
export function itemInformation(item: GRMItem, theta: number): number {
  const a = item.discrimination;
  let info = 0;
  for (let k = 0; k < item.numCategories; k++) {
    const pk = categoryProbability(item, theta, k);
    const pStarK = cumulativeProb(item, theta, k);
    const pStarK1 = cumulativeProb(item, theta, k + 1);
    // Derivative terms
    const dPstarK = pStarK * (1 - pStarK);
    const dPstarK1 = pStarK1 * (1 - pStarK1);
    const numerator = (dPstarK - dPstarK1) ** 2;
    info += numerator / pk;
  }
  return a * a * info;
}

/**
 * Total test information at θ for a set of items.
 */
export function testInformation(items: GRMItem[], theta: number): number {
  return items.reduce((sum, item) => sum + itemInformation(item, theta), 0);
}

// ── Theta Estimation ───────────────────────────────────────────

/**
 * Build quadrature grid for EAP estimation.
 * Uses equidistant points with normal prior weights.
 */
function buildQuadGrid(
  priorMean: number,
  priorSD: number,
  nPoints: number = QUAD_POINTS,
  range: number = QUAD_RANGE,
): QuadPoint[] {
  const points: QuadPoint[] = [];
  const lo = priorMean - range * priorSD;
  const hi = priorMean + range * priorSD;
  const step = (hi - lo) / (nPoints - 1);

  let totalWeight = 0;
  for (let i = 0; i < nPoints; i++) {
    const theta = lo + i * step;
    const z = (theta - priorMean) / priorSD;
    const weight = Math.exp(-0.5 * z * z);
    points.push({ theta, weight });
    totalWeight += weight;
  }
  // Normalize weights
  for (const p of points) {
    p.weight /= totalWeight;
  }
  return points;
}

/**
 * Expected A Posteriori (EAP) theta estimation.
 *
 * Computes posterior mean of θ given observed responses and a normal prior.
 * Uses Gauss-Hermite quadrature for numerical integration.
 *
 * Returns { theta, se } where se is the posterior standard deviation.
 */
export function estimateTheta(
  items: GRMItem[],
  responses: CATResponse[],
  priorMean: number = 0,
  priorSD: number = 1,
): { theta: number; se: number } {
  if (responses.length === 0) {
    return { theta: priorMean, se: priorSD };
  }

  const responseMap = new Map<string, number>();
  for (const r of responses) {
    responseMap.set(r.itemId, r.category);
  }

  const grid = buildQuadGrid(priorMean, priorSD);

  // Compute log-likelihood at each quadrature point
  const logLikelihoods: number[] = [];
  let maxLL = -Infinity;

  for (const point of grid) {
    let logL = 0;
    for (const item of items) {
      const cat = responseMap.get(item.id);
      if (cat === undefined) continue;
      const p = categoryProbability(item, point.theta, cat);
      logL += Math.log(p);
    }
    logLikelihoods.push(logL);
    if (logL > maxLL) maxLL = logL;
  }

  // Compute posterior (prior * likelihood) with log-sum-exp for stability
  let posteriorSum = 0;
  let thetaSum = 0;
  let thetaSqSum = 0;

  for (let i = 0; i < grid.length; i++) {
    const point = grid[i]!;
    const logL = logLikelihoods[i]!;
    // posterior ∝ prior_weight * exp(logL - maxLL)
    const posterior = point.weight * Math.exp(logL - maxLL);
    posteriorSum += posterior;
    thetaSum += point.theta * posterior;
    thetaSqSum += point.theta * point.theta * posterior;
  }

  if (posteriorSum === 0) {
    return { theta: priorMean, se: priorSD };
  }

  const thetaHat = thetaSum / posteriorSum;
  const variance = thetaSqSum / posteriorSum - thetaHat * thetaHat;
  const se = Math.sqrt(Math.max(0, variance));

  return { theta: thetaHat, se: se || priorSD };
}

// ── Utility ────────────────────────────────────────────────────

/**
 * Convert IRT theta to 0-100 percentile scale via normal CDF.
 * theta=0 → 50, theta=1 → ~84, theta=-1 → ~16.
 *
 * Uses Abramowitz & Stegun approximation (7-digit accuracy).
 */
export function thetaToPercentile(theta: number): number {
  if (theta < -6) return 0;
  if (theta > 6) return 100;

  const t = 1.0 / (1.0 + 0.2316419 * Math.abs(theta));
  const d = 0.3989422804014327; // 1/sqrt(2π)
  const p = d * Math.exp(-0.5 * theta * theta);
  const poly = t * (0.319381530 + t * (-0.356563782 + t * (1.781477937 + t * (-1.821255978 + t * 1.330274429))));
  const cdf = theta >= 0 ? 1.0 - p * poly : p * poly;
  return Math.round(cdf * 1000) / 10; // One decimal place
}

/**
 * Select the most informative item for a given theta from an item bank,
 * excluding already-administered items.
 */
export function selectNextItem(
  bank: GRMItem[],
  theta: number,
  administered: Set<string>,
): GRMItem | null {
  let bestItem: GRMItem | null = null;
  let bestInfo = -Infinity;

  for (const item of bank) {
    if (administered.has(item.id)) continue;
    const info = itemInformation(item, theta);
    if (info > bestInfo) {
      bestInfo = info;
      bestItem = item;
    }
  }
  return bestItem;
}

  /** Items administered for this dimension */
  administered: CATResponse[];
  /** Set of administered item IDs */
  administeredIds: Set<string>;
  /** Whether this dimension is complete */
  complete: boolean;
  /** Prior theta (from fixed-form if available) */
  priorMean: number;
  /** Prior SD (from fixed-form SE if available) */
  priorSD: number;
}

/** Full CAT session state. */
export interface CATSession {
  instrumentId: string;
  /** Per-dimension sessions */
  dimensions: Map<string, DimensionSession>;
  /** Order in which dimensions are tested */
  dimensionOrder: string[];
  /** Index of current dimension being tested */
  currentDimensionIndex: number;
  /** Total items administered across all dimensions */
  totalItemsAdministered: number;
  /** Whether the entire CAT is complete */
  complete: boolean;
}

/** Default stopping criteria for personality assessment. */
export const DEFAULT_STOPPING: StoppingCriteria = {
  seThreshold: 0.30,
  minItems: 3,
  maxItems: 10,
};

/**
 * Initialize a CAT session for a set of dimensions.
 *
 * @param instrumentId - Identifier for this CAT instrument
 * @param itemBank - Full item bank with GRM parameters
 * @param fixedFormScores - Optional scores from fixed-form instruments (theta estimates used as priors)
 * @param fixedFormItemIds - Item IDs from fixed-form instruments to exclude from CAT pool
 * @param stopping - Stopping criteria
 */
export function initCATSession(
  instrumentId: string,
  itemBank: GRMItem[],
  fixedFormScores?: ScaleScore[],
  fixedFormItemIds?: Set<string>,
  stopping: StoppingCriteria = DEFAULT_STOPPING,
): CATSession {
  // Group items by dimension
  const dimensionItems = new Map<string, GRMItem[]>();
  for (const item of itemBank) {
    // Skip items already administered in fixed form
    if (fixedFormItemIds?.has(item.id)) continue;
    const items = dimensionItems.get(item.dimensionId) ?? [];
    items.push(item);
    dimensionItems.set(item.dimensionId, items);
  }

  // Build dimension sessions
  const dimensions = new Map<string, DimensionSession>();
  const dimensionOrder: string[] = [];

  // Build fixed-form theta lookup
  const fixedTheta = new Map<string, { theta: number; se: number }>();
  if (fixedFormScores) {
    for (const score of fixedFormScores) {
      if (score.theta !== undefined && score.se !== undefined) {
        fixedTheta.set(score.scaleId, { theta: score.theta, se: score.se });
      }
    }
  }

  for (const [dimId] of dimensionItems) {
    // Skip dimension if fixed-form already meets SE threshold
    const prior = fixedTheta.get(dimId);
    if (prior && prior.se <= stopping.seThreshold) {
      continue; // Fixed form good enough, skip CAT for this facet
    }

    const priorMean = prior?.theta ?? 0;
    const priorSD = prior?.se ?? 1;

    dimensions.set(dimId, {
      dimensionId: dimId,
      theta: priorMean,
      se: priorSD,
      administered: [],
      administeredIds: new Set(),
      complete: false,
      priorMean,
      priorSD,
    });
    dimensionOrder.push(dimId);
  }

  return {
    instrumentId,
    dimensions,
    dimensionOrder,
    currentDimensionIndex: 0,
    totalItemsAdministered: 0,
    complete: dimensionOrder.length === 0,
  };
}

/**
 * Get the next item to administer.
 * Returns null if all dimensions are complete.
 */
export function getNextItem(
  session: CATSession,
  itemBank: GRMItem[],
): { item: GRMItem; dimensionId: string } | null {
  if (session.complete) return null;

  // Find next incomplete dimension
  while (session.currentDimensionIndex < session.dimensionOrder.length) {
    const dimId = session.dimensionOrder[session.currentDimensionIndex]!;
    const dimSession = session.dimensions.get(dimId);
    if (!dimSession) {
      session.currentDimensionIndex++;
      continue;
    }

    if (dimSession.complete) {
      session.currentDimensionIndex++;
      continue;
    }

    // Filter bank to this dimension
    const dimBank = itemBank.filter(
      (item) => item.dimensionId === dimId && !dimSession.administeredIds.has(item.id),
    );

    if (dimBank.length === 0) {
      dimSession.complete = true;
      session.currentDimensionIndex++;
      continue;
    }

    const nextItem = selectNextItem(dimBank, dimSession.theta, dimSession.administeredIds);
    if (!nextItem) {
      dimSession.complete = true;
      session.currentDimensionIndex++;
      continue;
    }

    return { item: nextItem, dimensionId: dimId };
  }

  // All dimensions complete
  session.complete = true;
  return null;
}

/**
 * Register a response and update theta estimate.
 */
export function registerResponse(
  session: CATSession,
  itemBank: GRMItem[],
  itemId: string,
  category: number,
  stopping: StoppingCriteria = DEFAULT_STOPPING,
): void {
  // Find which dimension this item belongs to
  const item = itemBank.find((i) => i.id === itemId);
  if (!item) return;

  const dimSession = session.dimensions.get(item.dimensionId);
  if (!dimSession) return;

  // Record response
  const response: CATResponse = { itemId, category };
  dimSession.administered.push(response);
  dimSession.administeredIds.add(itemId);
  session.totalItemsAdministered++;

  // Get all items for this dimension that have been administered
  const administeredItems = itemBank.filter(
    (i) => i.dimensionId === item.dimensionId && dimSession.administeredIds.has(i.id),
  );

  // Re-estimate theta
  const estimate = estimateTheta(
    administeredItems,
    dimSession.administered,
    dimSession.priorMean,
    dimSession.priorSD,
  );
  dimSession.theta = estimate.theta;
  dimSession.se = estimate.se;

  // Check stopping criteria
  const nItems = dimSession.administered.length;
  if (nItems >= stopping.maxItems) {
    dimSession.complete = true;
  } else if (nItems >= stopping.minItems && estimate.se <= stopping.seThreshold) {
    dimSession.complete = true;
  }

  // If current dimension complete, advance to next
  if (dimSession.complete) {
    session.currentDimensionIndex++;
    // Check if all dimensions are done
    if ([...session.dimensions.values()].every((d) => d.complete)) {
      session.complete = true;
    }
  }
}

/**
 * Get progress summary for all dimensions.
 */
export function getCATProgress(session: CATSession): {
  dimensionId: string;
  theta: number;
  se: number;
  itemsAdministered: number;
  complete: boolean;
  percentile: number;
}[] {
  return session.dimensionOrder.map((dimId) => {
    const dim = session.dimensions.get(dimId)!;
    return {
      dimensionId: dimId,
      theta: dim.theta,
      se: dim.se,
      itemsAdministered: dim.administered.length,
      complete: dim.complete,
      percentile: thetaToPercentile(dim.theta),
    };
  });
}

/**
 * Convert completed CAT session to ScaleScore array for storage.
 */
export function catSessionToScores(session: CATSession): ScaleScore[] {
  const scores: ScaleScore[] = [];
  for (const [dimId, dim] of session.dimensions) {
    scores.push({
      scaleId: dimId,
      scaleName: dimId, // Will be mapped to proper names by the instrument definition
      raw: dim.theta,
      normalized: thetaToPercentile(dim.theta),
      itemCount: dim.administered.length,
      theta: dim.theta,
      se: dim.se,
      itemsAdministered: dim.administered.length,
    });
  }
  return scores;
}

import { create } from "zustand";
import { persist } from "zustand/middleware";
import type {
  InstrumentSession,
  InstrumentResult,
  ItemResponse,
  Tier,
} from "../instruments/types";
import { getInstrument } from "../instruments/registry";

interface PsycheState {
  /** Active sessions keyed by instrument ID */
  sessions: Record<string, InstrumentSession>;
  /** Computed results keyed by instrument ID */
  results: Record<string, InstrumentResult>;
  /** Current instrument being taken */
  activeInstrumentId: string | null;
  /** Current item index within the active instrument */
  currentItemIndex: number;
  /** Selected tier for the instrument battery */
  selectedTier: Tier;

  // Actions
  setTier: (tier: Tier) => void;
  startInstrument: (instrumentId: string) => void;
  recordResponse: (response: ItemResponse) => void;
  advanceItem: () => void;
  goBackItem: () => void;
  completeInstrument: (result: InstrumentResult) => void;
  setActiveInstrument: (id: string | null) => void;
  resetInstrument: (instrumentId: string) => void;
  exportData: () => string;
  importData: (json: string) => void;
}

export const usePsycheStore = create<PsycheState>()(
  persist(
    (set, get) => ({
      sessions: {},
      results: {},
      activeInstrumentId: null,
      currentItemIndex: 0,
      selectedTier: "standard",

      setTier: (tier) => set(() => ({ selectedTier: tier })),

      startInstrument: (instrumentId) =>
        set((state) => ({
          activeInstrumentId: instrumentId,
          currentItemIndex: 0,
          sessions: {
            ...state.sessions,
            [instrumentId]: state.sessions[instrumentId] ?? {
              instrumentId,
              startedAt: Date.now(),
              responses: [],
            },
          },
        })),

      recordResponse: (response) =>
        set((state) => {
          const id = state.activeInstrumentId;
          if (!id) return state;
          const session = state.sessions[id];
          if (!session) return state;

          // Replace existing response for this item or append
          const existing = session.responses.findIndex(
            (r) => r.itemId === response.itemId
          );
          const responses = [...session.responses];
          if (existing >= 0) {
            responses[existing] = response;
          } else {
            responses.push(response);
          }

          return {
            sessions: {
              ...state.sessions,
              [id]: { ...session, responses },
            },
          };
        }),

      advanceItem: () =>
        set((state) => ({
          currentItemIndex: state.currentItemIndex + 1,
        })),

      goBackItem: () =>
        set((state) => ({
          currentItemIndex: Math.max(0, state.currentItemIndex - 1),
        })),

      completeInstrument: (result) =>
        set((state) => {
          const id = state.activeInstrumentId;
          if (!id) return state;
          const session = state.sessions[id];
          if (!session) return state;

          return {
            activeInstrumentId: null,
            currentItemIndex: 0,
            sessions: {
              ...state.sessions,
              [id]: { ...session, completedAt: Date.now() },
            },
            results: {
              ...state.results,
              [id]: result,
            },
          };
        }),

      setActiveInstrument: (id) =>
        set(() => ({
          activeInstrumentId: id,
          currentItemIndex: 0,
        })),

      resetInstrument: (instrumentId) =>
        set((state) => {
          const { [instrumentId]: _s, ...sessions } = state.sessions;
          const { [instrumentId]: _r, ...results } = state.results;
          return { sessions, results };
        }),

      exportData: () => {
        const { sessions, results, selectedTier } = get();
        return JSON.stringify({ sessions, results, selectedTier, exportedAt: Date.now() }, null, 2);
      },

      importData: (json) => {
        const data = JSON.parse(json);
        if (data.sessions && data.results) {
          // Merge with existing data (don't overwrite completed results)
          const currentState = get();
          const mergedSessions = { ...data.sessions, ...currentState.sessions };
          const mergedResults = { ...data.results, ...currentState.results };
          set({ sessions: mergedSessions, results: mergedResults });
          // Auto-score any sessions that have all responses but no result
          autoScoreCompleted();
        }
      },
    }),
    {
      name: "psyche-store",
      onRehydrateStorage: () => (state) => {
        // Always try to merge seed data (fills in any missing sessions/results)
        fetch("/seed-data.json")
          .then((r) => (r.ok ? r.json() : null))
          .then((data) => {
            if (data?.sessions) {
              const current = usePsycheStore.getState();
              const seedSessionCount = Object.keys(data.sessions).length;
              const currentSessionCount = Object.keys(current.sessions).length;
              if (seedSessionCount > currentSessionCount) {
                usePsycheStore.setState({
                  sessions: { ...data.sessions, ...current.sessions },
                  results: { ...(data.results ?? {}), ...current.results },
                });
              }
              autoScoreCompleted();
            }
          })
          .catch(() => {
            // No seed file or fetch failed — still auto-score existing data
            if (state) autoScoreCompleted();
          });
      },
    }
  )
);

/** Score any sessions that have all responses but no result yet. */
function autoScoreCompleted(): void {
  const { sessions, results } = usePsycheStore.getState();
  const newResults: Record<string, InstrumentResult> = {};

  for (const [id, session] of Object.entries(sessions)) {
    if (results[id]) continue; // already scored
    const registered = getInstrument(id);
    if (!registered) continue; // instrument not in registry
    const { instrument, score } = registered;
    if (session.responses.length >= instrument.items.length) {
      try {
        newResults[id] = score(instrument, session);
      } catch {
        // scoring failed — skip
      }
    }
  }

  if (Object.keys(newResults).length > 0) {
    const current = usePsycheStore.getState();
    usePsycheStore.setState({
      results: { ...current.results, ...newResults },
      sessions: Object.fromEntries(
        Object.entries(current.sessions).map(([id, s]) =>
          newResults[id] ? [id, { ...s, completedAt: s.completedAt ?? Date.now() }] : [id, s]
        )
      ),
    });
  }
}

        itemCount: 9,

/**

 * 60-item measure of Big Five personality with 30 facets (2 items per facet).
 * Subset of the Johnson 120-item IPIP-NEO pool (public domain).
 *
 * Item selection: first 2 items per facet from the 120-item pool.
 * Domain alpha ~0.72-0.78 (adequate for screening).
 * Facet-level reliability is low (~0.50-0.65) with only 2 items.
 *
 * Source: Maples-Keller, J. L., et al. (2019). Development of a 60-item form
 * of the IPIP NEO. Assessment.
 */

const DOMAINS: Record<string, { name: string; description: string }> = {
  N: { name: "Neuroticism", description: "Tendency to experience negative emotions such as anxiety, anger, and depression." },
  E: { name: "Extraversion", description: "Tendency to seek stimulation in the company of others and to experience positive emotions." },
  O: { name: "Openness", description: "Tendency to be imaginative, creative, and open to new experiences and ideas." },
  A: { name: "Agreeableness", description: "Tendency to be compassionate, cooperative, and trusting toward others." },
  C: { name: "Conscientiousness", description: "Tendency to be organized, dependable, disciplined, and goal-oriented." },
};

const FACETS: Record<string, string[]> = {
  N: ["Anxiety", "Anger", "Depression", "Self-Consciousness", "Immoderation", "Vulnerability"],
  E: ["Friendliness", "Gregariousness", "Assertiveness", "Activity Level", "Excitement-Seeking", "Cheerfulness"],
  O: ["Imagination", "Artistic Interests", "Emotionality", "Adventurousness", "Intellect", "Liberalism"],
  A: ["Trust", "Morality", "Altruism", "Cooperation", "Modesty", "Sympathy"],
  C: ["Self-Efficacy", "Orderliness", "Dutifulness", "Achievement-Striving", "Self-Discipline", "Cautiousness"],
};

function buildScales(): Scale[] {
  const scales: Scale[] = [];
  for (const [domainKey, meta] of Object.entries(DOMAINS)) {
    scales.push({ id: domainKey, name: meta.name, description: meta.description });
    const facetNames = FACETS[domainKey]!;
    for (let i = 0; i < facetNames.length; i++) {
      scales.push({ id: `${domainKey}${i + 1}`, name: facetNames[i]!, parentId: domainKey });
    }
  }
  return scales;
}

interface RawQuestion {
  id: string;
  text: string;
  keyed: "plus" | "minus";
  domain: string;
  facet: number;
}

/**
 * Select 2 items per facet from the 120-item pool.
 * Takes the first 2 items for each domain+facet combination.
 */
function buildItems(): Item[] {
  const all = questionsData as RawQuestion[];
  const selected: RawQuestion[] = [];

  for (const domain of Object.keys(DOMAINS)) {
    for (let facet = 1; facet <= 6; facet++) {
      const facetItems = all.filter((q) => q.domain === domain && q.facet === facet);
      selected.push(...facetItems.slice(0, 2));
    }
  }

  return selected.map((q) => ({
    id: q.id,
    text: q.text,
    response: {
      type: "likert" as const,
      min: 1,
      max: 5,
      labels: ["Very Inaccurate", "Moderately Inaccurate", "Neither", "Moderately Accurate", "Very Accurate"],
    },
    scaleId: `${q.domain}${q.facet}`,
    reversed: q.keyed === "minus",
  }));
}

const ipipNeo60: Instrument = {
  id: "ipip-neo-60",
  name: "IPIP-NEO-60",
  shortName: "Big Five (Brief)",
  description:

  citation:
    "Maples-Keller, J. L., Williamson, R. L., Sleep, C. E., Carter, N. T., Campbell, W. K., & Miller, J. D. (2019). Using item response theory to develop a 60-item representation of the NEO PI-R. Assessment, 26(1), 15-29.",
  itemCount: 60,
  estimatedMinutes: 8,
  scales: buildScales(),
  items: buildItems(),
};

registerInstrument(ipipNeo60, (instrument, session) => {
  const facetResult = scoreLikert(instrument, session);
  const domainScores = Object.keys(DOMAINS).map((domainKey) => {
    const facetScores = facetResult.scores.filter(
      (s) => s.scaleId.startsWith(domainKey) && s.scaleId.length > 1
    );
    const avgNorm = facetScores.length > 0
      ? facetScores.reduce((sum, s) => sum + s.normalized, 0) / facetScores.length
      : 0;
    const avgRaw = facetScores.length > 0
      ? facetScores.reduce((sum, s) => sum + s.raw, 0) / facetScores.length
      : 0;
    return {
      scaleId: domainKey,
      scaleName: DOMAINS[domainKey]!.name,
      raw: avgRaw,
      normalized: avgNorm,
      itemCount: facetScores.reduce((sum, s) => sum + s.itemCount, 0),
    };
  });
  return {
    instrumentId: instrument.id,
    completedAt: session.completedAt ?? Date.now(),
    scores: [...domainScores, ...facetResult.scores],
  };
});

      text: "Simon decided to invest $8,000 in the stock market one day early in 2008. Six months after he invested, on July 17, the stocks he had purchased were down 50%. Fortunately for Simon, from July 17 to October 17, the stocks he had purchased went up 75%. At this point, Simon has: (a) broken even in the stock market, (b) is ahead of where he began, (c) has lost money. Enter a, b, or c.",
      response: { type: "text" },
      scaleId: "crt-analytic",
    },
  ],
};

/** Correct answers for CRT-7 */
const CORRECT_ANSWERS: Record<string, (value: number | string) => boolean> = {
  "crt-1": (v) => Number(v) === 5,
  "crt-2": (v) => Number(v) === 5,
  "crt-3": (v) => Number(v) === 47,
  "crt-4": (v) => Number(v) === 4,
  "crt-5": (v) => Number(v) === 29,
  "crt-6": (v) => Number(v) === 20,
  "crt-7": (v) => String(v).toLowerCase().trim() === "c",
};

function scoreCrt(instrument: Instrument, session: InstrumentSession): InstrumentResult {
  let correct = 0;
  for (const resp of session.responses) {
    const checker = CORRECT_ANSWERS[resp.itemId];
    if (checker?.(resp.value)) {
      correct++;
    }
  }

  return {
    instrumentId: instrument.id,
    completedAt: session.completedAt ?? Date.now(),
    scores: [
      {
        scaleId: "crt-analytic",
        scaleName: "Analytic Thinking",
        raw: correct,
        normalized: (correct / 7) * 100,
        itemCount: session.responses.length,
      },
    ],
  };
}

    text,
    response: { type: "text" as const, minWords: 30 },

/** No scoring — just record completeness */
function scoreOpenEnded(_instrument: Instrument, session: InstrumentSession): InstrumentResult {
  const answered = session.responses.filter(
    (r) => typeof r.value === "string" && r.value.trim().length > 0
  ).length;

  const totalWords = session.responses.reduce((sum, r) => {
    if (typeof r.value === "string") {
      return sum + r.value.trim().split(/\s+/).length;
    }
    return sum;
  }, 0);

  return {
    instrumentId: "open-ended",
    completedAt: session.completedAt ?? Date.now(),
    scores: [
      {
        scaleId: "qualitative",
        scaleName: "Qualitative Responses",
        raw: answered,
        normalized: (answered / PROMPTS.length) * 100,
        itemCount: answered,
      },
      {
        scaleId: "word-count",
        scaleName: "Total Word Count",
        raw: totalWords,
        normalized: Math.min((totalWords / 1000) * 100, 100), // 1000 words = 100%
        itemCount: answered,
      },
    ],
  };
}

type Phase = 'loading' | 'consent' | 'instruments' | 'crisis-resources' | 'interview' | 'generating' | 'results' | 'error';

interface SessionState {
  sessionId: string;
  instrumentOrder: string[];
  currentInstrumentIndex: number;
  completedInstruments: Map<string, ItemResponse[]>;
  instrumentResults: InstrumentResult[];
  interviewResponses: Array<{ prompt_index: number; prompt_text: string; response_text: string }>;
  reportText?: string;
  reportLoading: boolean;
  reportError?: string;
  /** True if interview was already completed (e.g., via MCP) — skip website interview */
  interviewPreCompleted: boolean;
}

const STORAGE_KEY = 'psyche-session';
const STORAGE_VERSION = 2;
const SESSION_TTL_MS = 7 * 24 * 60 * 60 * 1000; // 7 days

interface StoredState {
  version: number;
  expiresAt: number;
  sessionId: string;
  tier: Tier;
  phase: Phase;
  instrumentIndex: number;
  completedIds: string[];
  instrumentResponses?: Record<string, ItemResponse[]>;
  instrumentResults?: InstrumentResult[];
  interviewResponses?: Array<{ prompt_index: number; prompt_text: string; response_text: string }>;
}

function saveToStorage(stored: Omit<StoredState, 'version' | 'expiresAt'> & { expiresAt?: number }) {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify({
      ...stored,
      version: STORAGE_VERSION,
      expiresAt: stored.expiresAt ?? Date.now() + SESSION_TTL_MS,
    }));
  } catch {}
}

function loadFromStorage(): StoredState | null {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return null;
    const stored = JSON.parse(raw) as StoredState;

    if (stored.version !== STORAGE_VERSION) {
      localStorage.removeItem(STORAGE_KEY);
      return null;
    }

    if (Date.now() > stored.expiresAt) {
      localStorage.removeItem(STORAGE_KEY);
      return null;
    }

    return stored;
  } catch {
    return null;
  }
}

function clearStorage() {
  try { localStorage.removeItem(STORAGE_KEY); } catch {}
}

export default function PsycheAssess() {
  const searchParams = useSearchParams();
  const [phase, setPhase] = useState<Phase>('loading');
  const [state, setState] = useState<SessionState>({
    sessionId: '',
    instrumentOrder: [],
    currentInstrumentIndex: 0,
    completedInstruments: new Map(),
    instrumentResults: [],
    interviewResponses: [],
    reportLoading: false,
    interviewPreCompleted: false,
  });
  const [error, setError] = useState<string>('');
  const [tier, setTier] = useState<Tier>('lite');
  const initRef = useRef(false);
  const expiresAtRef = useRef<number>(Date.now() + SESSION_TTL_MS);

  // Initialize session
  useEffect(() => {
    if (initRef.current) return;
    initRef.current = true;

    const init = async () => {
      // Check localStorage for resumed session to determine tier
      const stored = loadFromStorage();

      // Determine tier: stored session takes priority, then URL param, then default
      const urlTier = searchParams.get('tier');
      const resolvedTier: Tier = stored?.tier ?? (urlTier === 'standard' ? 'standard' : 'lite');
      setTier(resolvedTier);

      const instrumentIds = getInstrumentsForTier(resolvedTier);
      // Separate open-ended from quantitative instruments
      const quantitative = instrumentIds.filter(id => id !== 'open-ended');

      // Check for session param (MCP handoff)
      const sessionParam = searchParams.get('session');
      if (sessionParam) {
        try {
          const sessionData = await api.getSession(sessionParam);
          if (sessionData.error) {
            setError(sessionData.error);
            setPhase('error');
            return;
          }
          const hasInterview = !!sessionData.has_interview;

          // If interview exists (MCP flow), fetch it and store locally
          let interviewResponses: Array<{ prompt_index: number; prompt_text: string; response_text: string }> = [];
          if (hasInterview) {
            try {
              const exportData = await api.exportSession(sessionParam);
              if (exportData.interviews) {
                interviewResponses = exportData.interviews;
              }
            } catch {
              // Non-critical — interview data stays on server
            }
          }

          setState(prev => ({
            ...prev,
            sessionId: sessionParam,
            instrumentOrder: quantitative,
            interviewPreCompleted: hasInterview,
            interviewResponses,
          }));

          // Persist to localStorage so data survives refresh
          const expiresAt = Date.now() + SESSION_TTL_MS;
          expiresAtRef.current = expiresAt;
          saveToStorage({
            expiresAt,
            sessionId: sessionParam,
            tier: resolvedTier,
            phase: 'instruments',
            instrumentIndex: 0,
            completedIds: [],
            instrumentResponses: {},
            instrumentResults: sessionData.scores_json ? JSON.parse(JSON.stringify(sessionData.scores_json)) : [],
            interviewResponses,
          });

          // Resume based on session status
          if (sessionData.has_report) {
            setState(prev => ({
              ...prev,
              reportText: sessionData.report_text,
              instrumentResults: sessionData.scores_json ?? [],
            }));
            setPhase('results');
          } else if (sessionData.has_scores) {
            // Scores submitted — if interview pre-done, go to results; otherwise generating
            setPhase(hasInterview ? 'results' : 'generating');
          } else if (sessionData.consent_given >= 1) {
            setPhase('instruments');
          } else {
            setPhase('consent');
          }
          return;
        } catch {
          // Fall through to normal flow
        }
      }

      // Check localStorage for resumed session (stored already loaded at top of init)
      if (stored?.sessionId) {
        try {
          const sessionData = await api.getSession(stored.sessionId);
          if (!sessionData.error && sessionData.status === 'active') {
            // Rebuild completedInstruments Map from stored responses
            const completedMap = new Map<string, ItemResponse[]>();
            if (stored.instrumentResponses) {
              for (const [instId, resps] of Object.entries(stored.instrumentResponses)) {
                completedMap.set(instId, resps);
              }
            }

            const hasStoredInterview = (stored.interviewResponses?.length ?? 0) > 0;
            setState(prev => ({
              ...prev,
              sessionId: stored.sessionId,
              instrumentOrder: quantitative,
              currentInstrumentIndex: stored.instrumentIndex,
              completedInstruments: completedMap,
              instrumentResults: stored.instrumentResults ?? [],
              interviewResponses: stored.interviewResponses ?? [],
              interviewPreCompleted: hasStoredInterview && sessionData.source === 'mcp',
            }));

            // Preserve the TTL from the stored session
            expiresAtRef.current = stored.expiresAt;

            // If we have a report from the API, go to results
            if (sessionData.has_report) {
              setState(prev => ({
                ...prev,
                reportText: sessionData.report_text,
                instrumentResults: sessionData.scores_json ?? prev.instrumentResults,
              }));
              setPhase('results');
            } else {
              setPhase(stored.phase);
            }
            return;
          }
        } catch {
          // Session expired or invalid, start fresh
        }
      }

      setState(prev => ({
        ...prev,
        instrumentOrder: quantitative,
      }));
      setPhase('consent');
    };

    init();
  }, [searchParams]);

  // Handle consent
  const handleConsent = useCallback(async () => {
    try {
      const session = await api.createSession('website');
      if (session.error) {
        setError(session.error);
        setPhase('error');
        return;
      }
      await api.recordConsent(session.id, 1, CONSENT_TEXT_L1);
      setState(prev => ({ ...prev, sessionId: session.id }));
      const expiresAt = Date.now() + SESSION_TTL_MS;
      expiresAtRef.current = expiresAt;
      saveToStorage({ expiresAt, sessionId: session.id, tier, phase: 'instruments', instrumentIndex: 0, completedIds: [], instrumentResponses: {}, instrumentResults: [] });
      setPhase('instruments');
    } catch (err) {
      setError('Failed to create session. Please try again.');
      setPhase('error');
    }
  }, [tier]);

  // Handle instrument completion
  const handleInstrumentComplete = useCallback(async (responses: ItemResponse[]) => {
    const currentId = state.instrumentOrder[state.currentInstrumentIndex];
    if (!currentId) return;

    const registered = getInstrument(currentId);
    if (!registered) return;

    // Score locally
    const result = registered.score(registered.instrument, {
      instrumentId: currentId,
      startedAt: Date.now(),
      completedAt: Date.now(),
      responses,
    });

    // Save responses to API if research consent given
    const apiResponses = responses.map(r => ({
      instrument_id: currentId,
      item_id: r.itemId,
      value: r.value,
    }));

    // Fire and forget — don't block on response storage
    api.saveResponses(state.sessionId, apiResponses).catch(() => {});

    const newCompleted = new Map(state.completedInstruments);
    newCompleted.set(currentId, responses);

    const newResults = [...state.instrumentResults, result];
    const nextIndex = state.currentInstrumentIndex + 1;

    setState(prev => ({
      ...prev,
      completedInstruments: newCompleted,
      instrumentResults: newResults,
      currentInstrumentIndex: nextIndex,
    }));

    const completedIds = Array.from(newCompleted.keys());
    // Serialize Map to plain object for localStorage
    const storedResponses: Record<string, ItemResponse[]> = {};
    for (const [instId, resps] of newCompleted.entries()) {
      storedResponses[instId] = resps;
    }
    const allInstrumentsDone = nextIndex >= state.instrumentOrder.length;
    // Persist the actual phase that will render (including crisis interstitial)
    let nextPhase: Phase = allInstrumentsDone
      ? (state.interviewPreCompleted ? 'results' : 'interview')
      : 'instruments';
    if (!allInstrumentsDone && currentId === 'phq9-gad7') {
      nextPhase = 'crisis-resources';
    }

    saveToStorage({
      expiresAt: expiresAtRef.current,
      sessionId: state.sessionId,
      tier,
      phase: nextPhase,
      instrumentIndex: nextIndex,
      completedIds,
      instrumentResponses: storedResponses,
      instrumentResults: newResults,
    });

    if (allInstrumentsDone && state.interviewPreCompleted) {
      api.saveScores(state.sessionId, newResults).catch(() => {});
    }
    setPhase(nextPhase);
  }, [state, tier]);

  // Handle interview completion
  const handleInterviewComplete = useCallback(async (
    responses: Array<{ prompt_index: number; prompt_text: string; response_text: string }>
  ) => {
    setState(prev => ({ ...prev, interviewResponses: responses }));

    // Score open-ended once, build allResults locally
    const openEnded = getInstrument('open-ended');
    const allResults = [...state.instrumentResults];
    if (openEnded) {
      const oeResult = openEnded.score(openEnded.instrument, {
        instrumentId: 'open-ended',
        startedAt: Date.now(),
        completedAt: Date.now(),
        responses: responses.map((r, i) => ({
          itemId: `open-${i + 1}`,
          value: r.response_text,
          timestamp: Date.now(),
        })),
      });
      allResults.push(oeResult);
      setState(prev => ({
        ...prev,
        instrumentResults: [...prev.instrumentResults, oeResult],
      }));
    }

    // Persist to localStorage BEFORE network calls so data survives tab close
    const instrumentResponses: Record<string, ItemResponse[]> = {};
    for (const [instId, resps] of state.completedInstruments.entries()) {
      instrumentResponses[instId] = resps;
    }
    saveToStorage({
      expiresAt: expiresAtRef.current,
      sessionId: state.sessionId,
      tier,
      phase: 'results',
      instrumentIndex: state.currentInstrumentIndex,
      completedIds: Array.from(state.completedInstruments.keys()),
      instrumentResponses,
      instrumentResults: allResults,
      interviewResponses: responses,
    });

    try {
      await api.saveInterview(state.sessionId, responses);
      await api.saveScores(state.sessionId, allResults);
      setPhase('results');
    } catch {

      setError('Failed to save to server, but your data is saved locally.');
      setPhase('results');
    }
  }, [state, tier]);

        reportText: result.report_text,

      }));

      setState(prev => ({
        ...prev,
        reportLoading: false,
        reportError: 'Failed to generate report. Please try again.',
      }));
    }
  }, [state.sessionId]);

  // Handle research opt-in — upgrade consent then re-send all item responses
  const handleResearchOptIn = useCallback(async () => {

      const allResponses: Array<{ instrument_id: string; item_id: string; value: string | number }> = [];
      for (const [instId, resps] of state.completedInstruments.entries()) {
        for (const r of resps) {
          allResponses.push({
            instrument_id: instId,
            item_id: r.itemId,
            value: r.value,
          });
        }
      }
      if (allResponses.length > 0) {
        api.saveResponses(state.sessionId, allResponses).catch(() => {});
      }
    } catch {}
  }, [state.sessionId, state.completedInstruments]);

  // Build instruments list for progress display
  const allRegistered = getAllInstruments();
  const tierIds = getInstrumentsForTier(tier).filter(id => id !== 'open-ended');
  const instrumentsForProgress = tierIds
    .map(id => allRegistered.find(r => r.instrument.id === id)?.instrument)
    .filter(Boolean) as import('@/lib/psyche/types').Instrument[];

  // Current instrument
  const currentInstrumentId = state.instrumentOrder[state.currentInstrumentIndex];
  const currentRegistered = currentInstrumentId ? getInstrument(currentInstrumentId) : null;

  if (phase === 'loading') {
    return (
      <main className="max-w-4xl mx-auto px-4 py-12">
        <div className="flex items-center justify-center gap-3 text-[var(--color-text-muted)]">

import type {
  Instrument,
  InstrumentSession,
  InstrumentResult,
  ScaleScore,
  Item,
  ItemResponse,
} from "./types";

/**
 * Generic binary (True/False) scoring engine.
 * True = 1, False = 0; reverse scoring supported.
 */
export function scoreBinary(
  instrument: Instrument,
  session: InstrumentSession
): InstrumentResult {
  const responseMap = new Map<string, ItemResponse>();
  for (const r of session.responses) {
    responseMap.set(r.itemId, r);
  }

  const scaleItems = new Map<string, { item: Item; value: number }[]>();

  for (const item of instrument.items) {
    const resp = responseMap.get(item.id);
    if (!resp || typeof resp.value !== "number") continue;

    let value = resp.value;
    if (item.reversed) {
      value = 1 - value;
    }

    const existing = scaleItems.get(item.scaleId) ?? [];
    existing.push({ item, value });
    scaleItems.set(item.scaleId, existing);
  }

  const scores: ScaleScore[] = [];
  for (const scale of instrument.scales) {
    const items = scaleItems.get(scale.id);
    if (!items || items.length === 0) continue;

    const raw = items.reduce((sum, i) => sum + i.value, 0) / items.length;
    const normalized = raw * 100;

    scores.push({
      scaleId: scale.id,
      scaleName: scale.name,
      raw,
      normalized,
      itemCount: items.length,
    });
  }

  return {
    instrumentId: instrument.id,
    completedAt: session.completedAt ?? Date.now(),
    scores,
  };
}

/**
 * Generic Likert scoring engine.
 * Handles reverse scoring, scale aggregation, and normalization.
 */
export function scoreLikert(
  instrument: Instrument,
  session: InstrumentSession
): InstrumentResult {
  const responseMap = new Map<string, ItemResponse>();
  for (const r of session.responses) {
    responseMap.set(r.itemId, r);
  }

  const scaleItems = new Map<string, { item: Item; value: number }[]>();

  for (const item of instrument.items) {
    const resp = responseMap.get(item.id);
    if (!resp || typeof resp.value !== "number") continue;

    let value = resp.value;
    if (item.reversed && item.response.type === "likert") {
      const { min, max } = item.response;
      value = max + min - value;
    }

    const existing = scaleItems.get(item.scaleId) ?? [];
    existing.push({ item, value });
    scaleItems.set(item.scaleId, existing);
  }

  const scores: ScaleScore[] = [];
  for (const scale of instrument.scales) {
    const items = scaleItems.get(scale.id);
    if (!items || items.length === 0) continue;

    const raw = items.reduce((sum, i) => sum + i.value, 0) / items.length;

    const firstItem = instrument.items.find((i) => i.scaleId === scale.id);
    let normalized = raw;
    if (firstItem?.response.type === "likert") {
      const { min, max } = firstItem.response;
      normalized = ((raw - min) / (max - min)) * 100;
    }

    scores.push({
      scaleId: scale.id,
      scaleName: scale.name,
      raw,
      normalized,
      itemCount: items.length,
    });
  }

  return {
    instrumentId: instrument.id,
    completedAt: session.completedAt ?? Date.now(),
    scores,
  };
}

  instrument,
  existingResponses = [],

    () => {

      for (const r of existingResponses) {
        map.set(r.itemId, r);
      }
      return map;
    }
  );

  const [autoAdvance, setAutoAdvance] = useState(false);
  const pendingAdvanceRef = useRef(false);

  const items = instrument.items;
  const currentItem = items[currentIndex];
  const totalItems = items.length;

  // Report progress
  useEffect(() => {
    onProgress?.(responses.size, totalItems);
  }, [responses.size, totalItems, onProgress]);

  // Check completion
  useEffect(() => {
    if (responses.size === totalItems) {
      onComplete(Array.from(responses.values()));
    }
  }, [responses.size, totalItems, onComplete]);

  useEffect(() => {
    if (!pendingAdvanceRef.current) return;
    pendingAdvanceRef.current = false;

    if (currentIndex < totalItems - 1) {
      const timer = setTimeout(() => {
        setCurrentIndex(prev => prev + 1);
        setNumericInput('');

      return () => clearTimeout(timer);

              }}
              disabled={numericInput === ''}
              className="px-6 py-3 rounded-lg text-sm font-medium transition-all

                bg-[var(--color-primary)] text-[var(--color-bg)] hover:opacity-90"

            </button>

'use client';

import { useState, useCallback } from 'react';

const PROMPTS = [
  "Describe yourself as you would to a stranger who will never see you. Be as honest as possible.",
  "What are you most proud of? What are you least proud of?",
  "How do you typically respond to conflict or disagreement?",
  "Describe a decision you made recently that you found difficult. What made it hard?",
  "What gives you energy? What drains you?",
  "How have you changed in the last five years? What caused the change?",
  "What do your closest friends value most about you? What do they wish you'd change?",
  "Describe your relationship with uncertainty and ambiguity.",
  "What topic could you talk about for hours? Why does it matter to you?",
  "If you could design your ideal day with no obligations, what would it look like?",
];

interface InterviewResponse {
  prompt_index: number;
  prompt_text: string;
  response_text: string;
}

interface InterviewFlowProps {
  onComplete: (responses: InterviewResponse[]) => void;
  existingResponses?: InterviewResponse[];
}

export default function InterviewFlow({ onComplete, existingResponses = [] }: InterviewFlowProps) {
  const [currentIndex, setCurrentIndex] = useState(
    existingResponses.length > 0 ? Math.min(existingResponses.length, PROMPTS.length - 1) : 0
  );
  const [responses, setResponses] = useState<Map<number, string>>(() => {
    const map = new Map<number, string>();
    for (const r of existingResponses) {
      map.set(r.prompt_index, r.response_text);
    }
    return map;
  });
  const [currentText, setCurrentText] = useState(responses.get(currentIndex) ?? '');

  const wordCount = currentText.trim().split(/\s+/).filter(w => w.length > 0).length;
  const minWords = 30;
  const maxWords = 500;
  const isValid = wordCount >= minWords;

  const saveAndAdvance = useCallback(() => {
    if (!isValid) return;

    const next = new Map(responses);
    next.set(currentIndex, currentText.trim());
    setResponses(next);

    if (currentIndex < PROMPTS.length - 1) {
      const nextIdx = currentIndex + 1;
      setCurrentIndex(nextIdx);
      setCurrentText(next.get(nextIdx) ?? '');
    } else {
      // All done — collect responses
      const allResponses: InterviewResponse[] = [];
      for (let i = 0; i < PROMPTS.length; i++) {
        const text = i === currentIndex ? currentText.trim() : (next.get(i) ?? '');
        if (text) {
          allResponses.push({
            prompt_index: i,
            prompt_text: PROMPTS[i],
            response_text: text,
          });
        }
      }
      onComplete(allResponses);
    }
  }, [currentIndex, currentText, isValid, responses, onComplete]);

  const goBack = useCallback(() => {
    if (currentIndex > 0) {
      // Save current
      const next = new Map(responses);
      if (currentText.trim()) {
        next.set(currentIndex, currentText.trim());
      }
      setResponses(next);

      const prevIdx = currentIndex - 1;
      setCurrentIndex(prevIdx);
      setCurrentText(next.get(prevIdx) ?? '');
    }
  }, [currentIndex, currentText, responses]);

  const progressPct = ((responses.size + (currentText.trim() ? 0.5 : 0)) / PROMPTS.length) * 100;

  return (
    <div className="max-w-2xl mx-auto">
      {/* Progress */}
      <div className="mb-6">
        <div className="flex justify-between text-xs text-[var(--color-text-muted)] mb-1.5">
          <span>Interview</span>
          <span>{Math.min(responses.size + 1, PROMPTS.length)}/{PROMPTS.length}</span>
        </div>
        <div className="h-1.5 bg-[var(--color-surface)] rounded-full overflow-hidden">
          <div
            className="h-full rounded-full transition-all duration-300"
            style={{ width: `${progressPct}%`, background: 'var(--gradient-primary)' }}
          />
        </div>
      </div>

      {/* Prompt */}
      <div className="glass-card p-6 md:p-8 rounded-xl mb-4">
        <p className="text-xs text-[var(--color-text-muted)] mb-3">
          Question {currentIndex + 1} of {PROMPTS.length}
        </p>
        <p className="text-base md:text-lg text-[var(--color-text)] leading-relaxed mb-6">
          {PROMPTS[currentIndex]}
        </p>

        <textarea
          value={currentText}
          onChange={(e) => {
            // Soft limit at 500 words
            const words = e.target.value.trim().split(/\s+/).filter(w => w.length > 0).length;
            if (words <= maxWords + 50) { // Small buffer
              setCurrentText(e.target.value);
            }
          }}
          rows={6}
          className="w-full px-4 py-3 rounded-lg bg-[var(--color-surface)] border border-[var(--color-border)]
            text-[var(--color-text)] text-sm leading-relaxed resize-y
            focus:outline-none focus:border-[var(--color-primary)]
            placeholder:text-[var(--color-text-muted)]"
          placeholder="Write freely — there are no right answers..."
          autoFocus
        />

        <p className="text-[0.65rem] text-[var(--color-text-muted)] mt-2 opacity-70">
          For your privacy, please do not include your name, location, employer, or other
          personally identifying details in your responses.
        </p>

        <div className="flex justify-between items-center mt-2">
          <span className={`text-xs ${wordCount < minWords ? 'text-[var(--color-accent)]' : 'text-[var(--color-text-muted)]'}`}>
            {wordCount} / {minWords} words minimum
            {wordCount > maxWords && (
              <span className="text-[var(--color-accent)]"> (approaching limit)</span>
            )}
          </span>
        </div>
      </div>

      {/* Navigation */}
      <div className="flex justify-between items-center">
        <button
          onClick={goBack}
          disabled={currentIndex === 0}
          className="px-4 py-3 rounded-lg text-sm text-[var(--color-text-muted)] transition-colors
            disabled:opacity-30 hover:text-[var(--color-text)]"
        >
          Back
        </button>
        <button
          onClick={saveAndAdvance}
          disabled={!isValid}
          className="px-6 py-3 rounded-lg text-sm font-medium transition-all
            disabled:opacity-40 disabled:cursor-not-allowed
            bg-[var(--color-primary)] text-[var(--color-bg)] hover:opacity-90"
        >
          {currentIndex < PROMPTS.length - 1 ? 'Next' : 'Finish Interview'}
        </button>
      </div>
    </div>
  );
}

class ApiError extends Error {
  constructor(public status: number, message: string) {
    super(message);
    this.name = 'ApiError';
  }
}

async function fetchJson(url: string, options?: RequestInit) {
  const res = await fetch(url, options);
  const data = await res.json();
  if (!res.ok) {
    throw new ApiError(res.status, data.error || `HTTP ${res.status}`);
  }
  if (data.error) {
    throw new ApiError(res.status, data.error);
  }
  return data;
}

// --- Report Generation Prompt ---

const REPORT_PROMPT_VERSION = '1.0.0';

const REPORT_SYSTEM_PROMPT = `You are a personality assessment specialist. You will receive:
1. Psychometric instrument scores (normalized 0-100 per scale) from validated instruments
2. Free-text interview responses to 10 self-reflection prompts

Your task: Generate a comprehensive personality report that synthesizes quantitative scores with qualitative interview data.

Output format:
1. A JSON block with structured personality predictions (between <json> tags)
2. A ~1500-word narrative report (between <report> tags)

The JSON predictions should have this structure:
{
  "big_five": { "openness": 0-100, "conscientiousness": 0-100, "extraversion": 0-100, "agreeableness": 0-100, "neuroticism": 0-100 },
  "key_traits": ["trait1", "trait2", ...],
  "strengths": ["strength1", "strength2", ...],
  "growth_areas": ["area1", "area2", ...],
  "interview_themes": ["theme1", "theme2", ...]
}

The narrative report should:
- Integrate quantitative scores with qualitative evidence from interview responses
- Be written in second person ("You tend to...")
- Avoid clinical jargon; use accessible language
- Highlight patterns where multiple instruments converge on the same trait
- Note any interesting tensions between self-report scores and interview content
- Be honest but compassionate — acknowledge difficulties without pathologizing
- Include specific quotes or paraphrases from interview responses as evidence

IMPORTANT: This is NOT a clinical diagnosis. Frame all findings as personality patterns and tendencies, not disorders.

When discussing behavioral patterns:
- Describe coping strategies as learned or chosen behaviors, not as trauma responses. Even when someone describes difficult history, frame their current patterns as active strategies ("you tend to withdraw when overwhelmed") not nervous-system explanations ("your nervous system learned to stay on high alert"). The person is an agent making choices, not a body executing threat responses.
- Use hedged language for psychological constructs ("consistent with", "suggests a pattern of") rather than categorical labels ("you are anxious-preoccupied")
- Avoid pathologizing language in section headers — use descriptive framing ("How You Process Emotion") not clinical framing ("Anhedonic Depression", "Obsession")
- Stay in the lane of personality description. Do not recommend therapeutic techniques or treatment constructs (e.g. "earned secure attachment", "exposure work"). Growth areas should be framed as observations, not prescriptions.`;

function buildReportPrompt(scoresJson: string, interviews: { prompt_text: string; response_text: string }[]): string {
  let prompt = '## Psychometric Scores\n\n```json\n' + scoresJson + '\n```\n\n';
  prompt += '## Interview Responses\n\n';
  for (const interview of interviews) {
    prompt += `### Q: ${interview.prompt_text}\n\n${interview.response_text}\n\n`;
  }
  return prompt;
}

// --- Rate Limiting ---

const reportRateMap = new Map<string, number[]>();

function checkReportRateLimit(): boolean {
  const now = Date.now();
  const key = 'global';
  const timestamps = reportRateMap.get(key) ?? [];
  const recent = timestamps.filter(t => now - t < 3600000); // last hour
  reportRateMap.set(key, recent);
  return recent.length < 10;
}

function recordReportRequest(): void {
  const key = 'global';
  const timestamps = reportRateMap.get(key) ?? [];
  timestamps.push(Date.now());
  reportRateMap.set(key, timestamps);
}

function isSessionValid(session: Record<string, unknown>): boolean {
  if (session.status === 'withdrawn') return false;
  if (session.status === 'expired') return false;
  const expiresAt = new Date(session.expires_at as string);
  if (expiresAt < new Date()) return false;
  return true;
}

/** POST /api/psyche/sessions — Create anonymous session */
export async function createSession(request: Request, env: Env): Promise<Response> {
  const body = await request.json().catch(() => ({})) as Record<string, unknown>;
  const source = (body.source === 'mcp' ? 'mcp' : 'website') as string;

  const id = generateUUID();
  const now = new Date().toISOString();
  const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000).toISOString();

  await env.DB.prepare(
    `INSERT INTO psyche_sessions (id, created_at, updated_at, status, expires_at, source)
     VALUES (?, ?, ?, 'active', ?, ?)`
  ).bind(id, now, now, expiresAt, source).run();

  return jsonResponse({ id, status: 'active', expires_at: expiresAt, source }, 201);
}

/** GET /api/psyche/sessions/:id — Get session status */
export async function getSessionStatus(sessionId: string, env: Env): Promise<Response> {
  const session = await getSession(env.DB, sessionId);
  if (!session) return jsonResponse({ error: 'Session not found' }, 404);

  // Check expiry
  const expiresAt = new Date(session.expires_at as string);
  if (expiresAt < new Date() && session.status === 'active') {
    await env.DB.prepare(
      "UPDATE psyche_sessions SET status = 'expired', updated_at = datetime('now') WHERE id = ?"
    ).bind(sessionId).run();
    session.status = 'expired';
  }

  if (session.status === 'withdrawn') {
    return jsonResponse({ id: sessionId, status: 'withdrawn' }, 403);
  }

  // Check if interview responses exist
  const interviewCount = await env.DB.prepare(
    'SELECT COUNT(*) as count FROM psyche_interviews WHERE session_id = ?'
  ).bind(sessionId).first<{ count: number }>();

  return jsonResponse({
    id: session.id,
    status: session.status,
    consent_given: session.consent_given,
    expires_at: session.expires_at,
    source: session.source,
    has_scores: !!session.scores_json,
    has_report: !!session.report_text,
    has_interview: (interviewCount?.count ?? 0) > 0,
    scores_json: session.scores_json ? JSON.parse(session.scores_json as string) : null,
    report_text: session.report_text,
    report_model: session.report_model,
  });
}

/** POST /api/psyche/sessions/:id/consent — Record consent level */
export async function recordConsent(sessionId: string, request: Request, env: Env): Promise<Response> {
  const session = await getSession(env.DB, sessionId);
  if (!session) return jsonResponse({ error: 'Session not found' }, 404);
  if (!isSessionValid(session)) return jsonResponse({ error: 'Session expired or withdrawn' }, 410);

  const body = await request.json().catch(() => ({})) as Record<string, unknown>;
  const level = body.level as number;
  const consentText = body.consent_text as string;

  if (level !== 1 && level !== 2) {
    return jsonResponse({ error: 'Consent level must be 1 or 2' }, 400);
  }
  if (!consentText || typeof consentText !== 'string') {
    return jsonResponse({ error: 'consent_text is required' }, 400);
  }

  await env.DB.prepare(
    `UPDATE psyche_sessions
     SET consent_given = ?, consent_text = ?, consent_at = datetime('now'), updated_at = datetime('now')
     WHERE id = ?`
  ).bind(level, consentText, sessionId).run();

  return jsonResponse({ success: true, consent_given: level });
}

/** POST /api/psyche/sessions/:id/consent/withdraw — Withdraw consent */
export async function withdrawConsent(sessionId: string, env: Env): Promise<Response> {
  const session = await getSession(env.DB, sessionId);
  if (!session) return jsonResponse({ error: 'Session not found' }, 404);

  // Delete stored responses and interviews
  await env.DB.prepare('DELETE FROM psyche_responses WHERE session_id = ?').bind(sessionId).run();
  await env.DB.prepare('DELETE FROM psyche_interviews WHERE session_id = ?').bind(sessionId).run();

  await env.DB.prepare(
    `UPDATE psyche_sessions
     SET status = 'withdrawn', withdrawal_at = datetime('now'), updated_at = datetime('now')
     WHERE id = ?`
  ).bind(sessionId).run();

  return jsonResponse({ success: true, status: 'withdrawn' });
}

/** POST /api/psyche/sessions/:id/responses — Batch save instrument responses */
export async function saveResponses(sessionId: string, request: Request, env: Env): Promise<Response> {
  const session = await getSession(env.DB, sessionId);
  if (!session) return jsonResponse({ error: 'Session not found' }, 404);
  if (!isSessionValid(session)) return jsonResponse({ error: 'Session expired or withdrawn' }, 410);

  // Only store if consent >= 2 (research)
  if ((session.consent_given as number) < 2) {
    return jsonResponse({ success: true, stored: false, reason: 'Research consent not given' });
  }

  const body = await request.json().catch(() => ({})) as Record<string, unknown>;
  const responses = body.responses as Array<{ instrument_id: string; item_id: string; value: string | number }>;

  if (!Array.isArray(responses) || responses.length === 0) {
    return jsonResponse({ error: 'responses array is required' }, 400);
  }

  // Batch insert
  const stmt = env.DB.prepare(
    'INSERT INTO psyche_responses (session_id, instrument_id, item_id, value) VALUES (?, ?, ?, ?)'
  );
  const batch = responses.map(r =>
    stmt.bind(sessionId, r.instrument_id, r.item_id, String(r.value))
  );

  await env.DB.batch(batch);

  return jsonResponse({ success: true, stored: true, count: responses.length });
}

/** POST /api/psyche/sessions/:id/interview — Save interview responses */
export async function saveInterview(sessionId: string, request: Request, env: Env): Promise<Response> {
  const session = await getSession(env.DB, sessionId);
  if (!session) return jsonResponse({ error: 'Session not found' }, 404);
  if (!isSessionValid(session)) return jsonResponse({ error: 'Session expired or withdrawn' }, 410);

  const rawBody = await request.text();
  if (rawBody.length > 51200) { // 50KB limit
    return jsonResponse({ error: 'Request body too large (max 50KB)' }, 413);
  }

  const body = JSON.parse(rawBody) as Record<string, unknown>;
  const interviews = body.interviews as Array<{ prompt_index: number; prompt_text: string; response_text: string }>;

  if (!Array.isArray(interviews) || interviews.length === 0) {
    return jsonResponse({ error: 'interviews array is required' }, 400);
  }

  if (interviews.length > 10) {
    return jsonResponse({ error: 'Maximum 10 interview responses' }, 400);
  }

  // Validate each interview
  for (const interview of interviews) {
    if (typeof interview.prompt_index !== 'number' || !interview.prompt_text || !interview.response_text) {
      return jsonResponse({ error: 'Each interview must have prompt_index, prompt_text, and response_text' }, 400);
    }
    const wordCount = interview.response_text.trim().split(/\s+/).length;
    if (wordCount < 10) {
      return jsonResponse({ error: `Interview response ${interview.prompt_index} is too short (min 10 words)` }, 400);
    }
  }

  const stmt = env.DB.prepare(
    'INSERT INTO psyche_interviews (session_id, prompt_index, prompt_text, response_text, word_count) VALUES (?, ?, ?, ?, ?)'
  );

  const batch = interviews.map(i => {
    const wordCount = i.response_text.trim().split(/\s+/).length;
    return stmt.bind(sessionId, i.prompt_index, i.prompt_text, i.response_text, wordCount);
  });

  await env.DB.batch(batch);

  return jsonResponse({ success: true, count: interviews.length });
}

/** POST /api/psyche/sessions/:id/report — Generate narrative report via Kimi K2.5 */
export async function generateReport(sessionId: string, env: Env): Promise<Response> {
  const session = await getSession(env.DB, sessionId);
  if (!session) return jsonResponse({ error: 'Session not found' }, 404);
  if (!isSessionValid(session)) return jsonResponse({ error: 'Session expired or withdrawn' }, 410);

  // Only 1 report per session
  if (session.report_text) {
    return jsonResponse({ error: 'Report already generated for this session' }, 409);
  }

  // Check scores exist
  if (!session.scores_json) {
    return jsonResponse({ error: 'Scores must be saved before generating a report' }, 400);
  }

  // Global rate limit
  if (!checkReportRateLimit()) {
    return jsonResponse(
      { error: 'Report generation rate limit exceeded. Try again later.' },
      429,
    );
  }

  // Fetch interview responses
  const interviews = await env.DB.prepare(
    'SELECT prompt_text, response_text FROM psyche_interviews WHERE session_id = ? ORDER BY prompt_index'
  ).bind(sessionId).all();

  if (!interviews.results || interviews.results.length === 0) {
    return jsonResponse({ error: 'Interview responses must be saved before generating a report' }, 400);
  }

  const userPrompt = buildReportPrompt(
    session.scores_json as string,
    interviews.results as unknown as { prompt_text: string; response_text: string }[]
  );

  recordReportRequest();

  const startTime = Date.now();

  // Call DeepInfra (OpenAI-compatible endpoint)
  const aiResponse = await fetch('https://api.deepinfra.com/v1/openai/chat/completions', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${env.DEEPINFRA_API_KEY}`,
    },
    body: JSON.stringify({
      model: 'moonshotai/Kimi-K2.5',
      messages: [
        { role: 'system', content: REPORT_SYSTEM_PROMPT },
        { role: 'user', content: userPrompt },
      ],
      max_tokens: 4096,
      temperature: 0.3,
    }),
  });

  if (!aiResponse.ok) {
    const errText = await aiResponse.text().catch(() => 'Unknown error');
    console.error('DeepInfra error:', aiResponse.status, errText);
    return jsonResponse(
      { error: 'Report generation temporarily unavailable. Try again later.' },
      503,
    );
  }

  const aiData = await aiResponse.json() as {
    choices: Array<{ message: { content: string } }>;
    usage?: { prompt_tokens?: number; completion_tokens?: number };
  };

  const latencyMs = Date.now() - startTime;
  const reportContent = aiData.choices?.[0]?.message?.content ?? '';
  const tokensIn = aiData.usage?.prompt_tokens ?? 0;
  const tokensOut = aiData.usage?.completion_tokens ?? 0;

  // Store report in session
  await env.DB.prepare(
    `UPDATE psyche_sessions
     SET report_text = ?, report_model = 'kimi-k2.5', status = 'completed', updated_at = datetime('now')
     WHERE id = ?`
  ).bind(reportContent, sessionId).run();

  // Store evaluation for benchmarking
  // Extract JSON predictions from report
  let evaluationJson = '{}';
  const jsonMatch = reportContent.match(/<json>([\s\S]*?)<\/json>/);
  if (jsonMatch) {
    evaluationJson = jsonMatch[1].trim();
  }

  await env.DB.prepare(
    `INSERT INTO psyche_evaluations (session_id, model_id, model_provider, prompt_version, evaluation_json, tokens_in, tokens_out, latency_ms)
     VALUES (?, 'kimi-k2.5', 'deepinfra', ?, ?, ?, ?, ?)`
  ).bind(sessionId, REPORT_PROMPT_VERSION, evaluationJson, tokensIn, tokensOut, latencyMs).run();

  return jsonResponse({
    success: true,
    report_text: reportContent,
    report_model: 'kimi-k2.5',
    tokens_in: tokensIn,
    tokens_out: tokensOut,
    latency_ms: latencyMs,
  });
}

/** GET /api/psyche/sessions/:id/export — Export user's own data */

  const interviews = await env.DB.prepare(

    report: session.report_text,
    report_model: session.report_model,
    interviews: interviews.results,
  });
}

/** POST /api/psyche/cleanup — Delete expired sessions */
export async function cleanupSessions(request: Request, env: Env): Promise<Response> {
  const auth = request.headers.get('Authorization');
  if (!auth || auth !== `Bearer ${env.PLAYGROUND_TOKEN}`) {
    return jsonResponse({ error: 'Unauthorized' }, 401);
  }

  // Delete expired active sessions (keep completed ones with research consent)
  const result = await env.DB.prepare(
    `DELETE FROM psyche_sessions
     WHERE expires_at < datetime('now')
     AND status = 'active'`
  ).run();

  // Also clean up withdrawn sessions older than 24 hours
  await env.DB.prepare(
    `DELETE FROM psyche_sessions
     WHERE status = 'withdrawn'
     AND withdrawal_at < datetime('now', '-1 day')`
  ).run();

  return jsonResponse({
    success: true,
    deleted: result.meta?.changes ?? 0,
  });
}
